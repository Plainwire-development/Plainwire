cmake_minimum_required(VERSION 3.16)
project(plainwire_bot_cpp VERSION 2.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(../c ${CMAKE_CURRENT_BINARY_DIR}/plainwire-c)
add_library(plainwire_bot_cpp INTERFACE)
target_include_directories(plainwire_bot_cpp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(plainwire_bot_cpp INTERFACE plainwire_bot)
add_executable(plainwire_cpp_echo examples/echo_bot.cpp)
target_link_libraries(plainwire_cpp_echo PRIVATE plainwire_bot_cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(plainwire_cpp_echo PRIVATE -Wall -Wextra -Wpedantic)
endif()
