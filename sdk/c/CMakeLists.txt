cmake_minimum_required(VERSION 3.16)
project(plainwire_bot_c VERSION 2.2.0 LANGUAGES C)
find_package(CURL REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(CURL_PC QUIET IMPORTED_TARGET libcurl)
endif()
add_library(plainwire_bot STATIC src/plainwire_bot.c)
target_include_directories(plainwire_bot PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(TARGET PkgConfig::CURL_PC)
  # Some environments expose libcurl only as a static archive. The imported
  # pkg-config target carries the required transitive libraries in that case.
  target_link_libraries(plainwire_bot PUBLIC PkgConfig::CURL_PC)
else()
  target_link_libraries(plainwire_bot PUBLIC CURL::libcurl)
endif()
target_compile_features(plainwire_bot PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(plainwire_bot PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(plainwire_echo examples/echo_bot.c)
target_link_libraries(plainwire_echo PRIVATE plainwire_bot)

include(CTest)
if(BUILD_TESTING)
  add_executable(plainwire_c_policy_test tests/policy_test.c)
  target_link_libraries(plainwire_c_policy_test PRIVATE plainwire_bot)
  add_test(NAME plainwire_c_policy COMMAND plainwire_c_policy_test)
endif()
